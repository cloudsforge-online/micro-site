# CloudsForge — the marketing site, and everything else on the apex.
#
# Nothing on the marketing site itself sits behind a session, so it has nothing of its own to
# disallow. The one rule below belongs to a DIFFERENT bundle: Forge Journal is served at /journal on
# this origin, and a crawler reads robots.txt at the origin root and nowhere else, so the rules for
# every surface mounted on the apex have to be stated here or they do not exist anywhere.
#
# THIS FILE IS THE DEVELOPMENT COPY. In production nginx serves /robots.txt itself, from a block in
# nginx.conf, because the `Sitemap:` lines have to carry ABSOLUTE URLs and no artefact built in
# this repository is allowed to name a hostname — the same image is served from localhost, from the
# testnet host and from the apex. nginx derives the host from the request with `$host`, which it can
# and this file cannot. (The SCHEME is a literal `https` there rather than `$scheme`, because TLS
# ends at Cloudflare and `$scheme` is `http` at the origin for a reader who arrived over https.)
#
# The two copies are compared rather than trusted: test/routes.test.ts reads the rules out of both
# and fails if they disagree, so only the Sitemap lines may ever differ.

User-agent: *
Allow: /

# Forge Journal's search is a client-side pass over the whole corpus: every `?q=` is a separate
# address rendering the same shell, so it is crawl budget spent on something that is not a document.
# The articles themselves are in the journal's own sitemap, which nginx announces alongside this
# site's — per article and per topic, each with a real `lastmod` from the article's front matter.
Disallow: /journal/search

# Forge Agora is served at /agora on this origin, and these six are ITS rules, not this site's —
# moved here in wave 3c of the consolidation when the square stopped having a robots.txt of its own.
#
# Five are the reader's own pages. Each needs a token and answers 401 without one, so a crawler
# gains nothing by fetching them; but `/agora/whispers` in a search index is a URL that READS like a
# public page of private messages, and that impression is worth six lines to avoid. `/agora/search`
# is public and disallowed for the ordinary reason: an infinite query space is a full-text scan of
# the whole square, once per crawler.
#
# `/agora/moderation` is private and deliberately absent. Nothing links to it, its own robots meta
# says noindex, and a Disallow is a PUBLIC statement that an address exists — naming the operator
# queue in the first file a curious stranger opens is worse than the risk it would mitigate.
#
# These six are DERIVED, in the served copy: they come from `noIndexPaths` on agora's row in
# micro-ui's surface registry, prefixed with its basePath, and test/routes.test.ts fails if any
# declared path is missing from nginx.conf. This development copy is compared against that one, so
# it cannot drift either.
Disallow: /agora/home
Disallow: /agora/notifications
Disallow: /agora/whispers
Disallow: /agora/bookmarks
Disallow: /agora/settings
Disallow: /agora/search

# usually returns, so a crawler that follows a stale link is told the truth and can drop it.
